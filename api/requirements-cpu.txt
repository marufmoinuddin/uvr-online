# UVR Local CPU worker — Python 3.10
#
# The CPU counterpart of requirements.txt: same application, same model
# catalogue, same API — but torch from the CPU wheel index and no CUDA
# runtime at all. It is what the `cpu` compose profile and `run-local.sh
# --cpu` use.
#
# Kept as a separate file rather than a build argument on requirements.txt
# because the difference is not just a URL: the CPU image also drops the
# onnxruntime-gpu wheel for the much smaller CPU one, and pinning torch to
# the `+cpu` local version is what keeps pip from quietly resolving the CUDA
# build from PyPI.

# --- PyTorch (CPU) ---
# Index options MUST be on their own line: pip ignores an inline
# `--index-url`, then looks for +cpu builds on PyPI, which has none, and the
# build fails. Use --extra-index-url (not --index-url) so PyPI stays
# available for the non-torch packages below.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.8.0+cpu
torchvision==0.23.0+cpu
torchaudio==2.8.0+cpu

# --- API ---
fastapi==0.115.0
uvicorn[standard]==0.34.0
python-multipart==0.0.12
websockets==13.1
python-dotenv==1.0.1

# --- Audio / DSP ---
# audio-separator requires numpy>=2 and librosa>=0.10, so these are no longer
# held back at the older 1.x / 0.9 releases.
numpy>=2,<3
scipy>=1.11
soundfile>=0.12
librosa>=0.10,<0.12
safetensors>=0.4

# --- Inference (standalone) ---
# The `cpu` extra pulls onnxruntime (CPU). The `gpu` extra pulls
# onnxruntime-gpu, which is several hundred megabytes of CUDA libraries for
# nothing on a machine with no GPU.
audio-separator[cpu]>=0.40,<0.50

# --- Bundled MSST inference (HyperACE etc.) ---
# api/vendor/msst carries MSST-WebUI's own BS-RoFormer implementation so the
# app can run checkpoints audio-separator cannot (notably the HyperACE
# variants, which use FNO layers). Requires neuraloperator for those layers.
neuraloperator>=2.0.0
beartype>=0.18.5
rotary-embedding-torch>=0.6.1
